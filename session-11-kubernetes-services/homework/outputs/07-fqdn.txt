### /etc/resolv.conf in each pod - the search list depends on the pod's namespace
$ kubectl exec dnsutils -- cat /etc/resolv.conf
search default.svc.cluster.local svc.cluster.local cluster.local
nameserver 10.96.0.10
options ndots:5

$ kubectl exec -n team-b dnsutils -- cat /etc/resolv.conf
search team-b.svc.cluster.local svc.cluster.local cluster.local
nameserver 10.96.0.10
options ndots:5

### From namespace default: the short name 'api' does NOT find team-b's Service
$ kubectl exec dnsutils -- nslookup api
Server:		10.96.0.10
Address:	10.96.0.10#53

** server can't find api: SERVFAIL

command terminated with exit code 1

### From namespace default: <svc>.<ns>, <svc>.<ns>.svc and the full FQDN all work
$ kubectl exec dnsutils -- nslookup api.team-b

Address:	10.96.0.10#53

Address: 10.96.101.126

### Pod-to-service communication across namespaces (HTTP)
$ kubectl exec curl-client -- curl -s -o /dev/null -w 'default -> http://api.team-b.svc.cluster.local -> HTTP %{http_code} via %{remote_ip}\n' http://api.team-b.svc.cluster.local
default -> http://api.team-b.svc.cluster.local -> HTTP 200 via 10.96.101.126

$ kubectl exec curl-client -- curl -s -m 5 -o /dev/null -w 'default -> http://api -> HTTP %{http_code}\n' http://api
default -> http://api -> HTTP 000
command terminated with exit code 28

### How the search list expands a short name (dig +search shows what finally matched)
$ kubectl exec dnsutils -- dig +search +noall +answer api.team-b
api.team-b.svc.cluster.local. 30 IN	A	10.96.101.126

### A trailing dot makes the name absolute: no search list, one single query
$ kubectl exec dnsutils -- nslookup api.team-b.svc.cluster.local.
Server:		10.96.0.10
Address:	10.96.0.10#53

Name:	api.team-b.svc.cluster.local
Address: 10.96.101.126
