# The instructor's target domain does not exist any more (NXDOMAIN). Kubernetes never checks the externalName -
# the Service was created fine and the CNAME is served, it just leads nowhere. So I added two working ones:
$ cat services/04-externalname/service-extra.yaml
# Two extra ExternalName Services I added, because the instructor's target (nencyravaliya.me)
# no longer resolves (NXDOMAIN) when I ran this on 2026-10-07.
# 1) an alias to a public website
apiVersion: v1
kind: Service
metadata:
  name: external-web
spec:
  type: ExternalName
  externalName: example.com
---
# 2) an alias to a Service in another namespace - the most common real use
#    (apps in "default" keep calling "backend-alias"; the real backend lives elsewhere)
apiVersion: v1
kind: Service
metadata:
  name: backend-alias
spec:
  type: ExternalName
  externalName: web-service-clusterip.default.svc.cluster.local

$ kubectl apply -f services/04-externalname/service-extra.yaml
service/external-web unchanged
service/backend-alias unchanged

$ kubectl get svc external-web backend-alias
NAME            TYPE           CLUSTER-IP   EXTERNAL-IP                                       PORT(S)   AGE
external-web    ExternalName   <none>       example.com                                       <none>    4s
backend-alias   ExternalName   <none>       web-service-clusterip.default.svc.cluster.local   <none>    4s

### external-web -> example.com
$ kubectl exec dnsutils -- dig +noall +answer external-web.default.svc.cluster.local
external-web.default.svc.cluster.local.	30 IN CNAME example.com.
example.com.		30	IN	A	104.20.23.154
example.com.		30	IN	A	172.66.147.243

$ kubectl exec dns-test-client -- curl -s -m 10 -o /dev/null -w 'http://external-web -> HTTP %{http_code} from %{remote_ip}\n' http://external-web
http://external-web -> HTTP 403 from 104.20.23.154

$ kubectl exec dns-test-client -- curl -s -m 10 -o /dev/null -w 'http://external-web with Host: example.com -> HTTP %{http_code} from %{remote_ip}\n' -H 'Host: example.com' http://external-web
http://external-web with Host: example.com -> HTTP 200 from 104.20.23.154

# Same IP both times. Only the DNS name is aliased; HTTP still sends 'Host: external-web', and the CDN in front of
# example.com refuses an unknown host (403). With the right Host header it answers 200.
### backend-alias -> web-service-clusterip.default.svc.cluster.local (in-cluster alias)
