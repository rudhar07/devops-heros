$ cat services/03-loadbalancer/metallb-pool.yaml
# MetalLB = the "cloud load balancer" for my kind cluster.
# The pool is a small slice of the docker "kind" network (172.18.0.0/16), far away from the
# addresses Docker hands out to node containers (172.18.0.x), so it cannot collide with them.
apiVersion: metallb.io/v1beta1
kind: IPAddressPool
metadata:
  name: hw-a-pool
  namespace: metallb-system
spec:
  addresses:
    - 172.18.203.200-172.18.203.210
---
apiVersion: metallb.io/v1beta1
kind: L2Advertisement
metadata:
  name: hw-a-l2
  namespace: metallb-system
spec:
  ipAddressPools:
    - hw-a-pool

# (my script first applied the pool in a silent retry loop, because MetalLB's admission webhook needs a few
#  seconds after the pods are Ready; this visible apply therefore reports 'unchanged')
$ kubectl apply -f services/03-loadbalancer/metallb-pool.yaml
ipaddresspool.metallb.io/hw-a-pool unchanged
l2advertisement.metallb.io/hw-a-l2 unchanged

### Test connectivity to the EXTERNAL-IP 172.18.203.200
# a) from a container on the docker 'kind' network (same L2 segment MetalLB answers ARP on):
$ docker run --rm --network kind curlimages/curl:8.5.0 -s -o /dev/null -w 'kind network -> http://172.18.203.200 -> HTTP %{http_code}\n' http://172.18.203.200
kind network -> http://172.18.203.200 -> HTTP 200

$ docker run --rm --network kind --entrypoint sh curlimages/curl:8.5.0 -c "curl -s http://172.18.203.200 | grep -o '<title>.*</title>'"
<title>Welcome to nginx!</title>

# b) from a pod inside the cluster:
