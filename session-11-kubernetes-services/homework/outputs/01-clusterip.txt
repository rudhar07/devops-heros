### Test connectivity from inside the cluster (short name, FQDN, ClusterIP)
$ kubectl exec curl-client -- curl -s -o /dev/null -w 'short name -> HTTP %{http_code} from %{remote_ip}:%{remote_port}\n' http://web-service-clusterip:8080
short name -> HTTP 200 from 10.96.96.222:8080

$ kubectl exec curl-client -- curl -s -o /dev/null -w 'FQDN       -> HTTP %{http_code} from %{remote_ip}:%{remote_port}\n' http://web-service-clusterip.default.svc.cluster.local:8080
FQDN       -> HTTP 200 from 10.96.96.222:8080

$ kubectl exec curl-client -- curl -s -o /dev/null -w 'ClusterIP  -> HTTP %{http_code} from %{remote_ip}:%{remote_port}\n' http://10.96.96.222:8080
ClusterIP  -> HTTP 200 from 10.96.96.222:8080

$ kubectl exec curl-client -- sh -c 'curl -s http://web-service-clusterip:8080 | grep -o "<title>.*</title>"'
<title>Welcome to nginx!</title>

### Load balancing: 30 requests, then count access-log lines per pod
$ kubectl exec curl-client -- sh -c 'for i in $(seq 1 30); do curl -s -o /dev/null http://web-service-clusterip:8080/lb-test; done; echo sent 30'
sent 30

$ for p in $(kubectl get pods -l app=web-clusterip -o name); do echo "$p $(kubectl logs $p | grep -c "GET /lb-test HTTP/1.1\" 404")"; done    # access-log lines only
pod/web-app-clusterip-66865d4855-2w84d 10
pod/web-app-clusterip-66865d4855-fb4sc 11
pod/web-app-clusterip-66865d4855-j24pn 9
