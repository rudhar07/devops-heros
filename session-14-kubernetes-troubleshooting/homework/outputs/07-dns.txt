# ===== DNS 3: short name from another namespace =====
# A pod in namespace 'production' calls 'web-service' (which lives in 'default'):
$ kubectl -n production run dns-client --image=busybox:1.36 --restart=Never --command -- sleep 600
pod/dns-client created

$ kubectl -n production wait --for=condition=Ready pod/dns-client --timeout=120s
pod/dns-client condition met

$ kubectl -n production exec dns-client -- cat /etc/resolv.conf
search production.svc.cluster.local svc.cluster.local cluster.local
nameserver 10.96.0.10
options ndots:5

$ kubectl -n production exec dns-client -- nslookup web-service
Server:		10.96.0.10
Address:	10.96.0.10:53

** server can't find web-service.production.svc.cluster.local: NXDOMAIN

** server can't find web-service.cluster.local: NXDOMAIN

** server can't find web-service.svc.cluster.local: NXDOMAIN

** server can't find web-service.cluster.local: NXDOMAIN

# ===== busybox nslookup quirk (found while doing the above) =====
# Above, 'nslookup kubernetes.default' and 'nslookup web-service.default' said NXDOMAIN, but 'wget http://web-service.default' worked.
# busybox's nslookup does not apply the search list to names that already contain a dot. The libc resolver used by wget does.
# So: don't trust busybox nslookup for partial names. Use the FQDN, or test with the real client (wget/curl/getent).
$ kubectl exec dns-test -- nslookup kubernetes.default.svc.cluster.local
Server:		10.96.0.10
Address:	10.96.0.10:53


Name:	kubernetes.default.svc.cluster.local
Address: 10.96.0.1
