# Mitigation on MY single-node cluster: turn off leader election for the scheduler and controller-manager.
# With one replica each there is nobody to fail over to, and the lease renewals were timing out because etcd was starved.
# (Edits the static-pod manifests inside the kind node container; kubelet restarts the pods by itself.)
$ docker exec hw-b-control-plane sed -i 's/--leader-elect=true/--leader-elect=false/' /etc/kubernetes/manifests/kube-scheduler.yaml

$ docker exec hw-b-control-plane sed -i 's/--leader-elect=true/--leader-elect=false/' /etc/kubernetes/manifests/kube-controller-manager.yaml
