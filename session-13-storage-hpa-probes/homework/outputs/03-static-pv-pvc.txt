# Step 2: my copy sets storageClassName: "" (and volumeName) so it binds to the static PV
$ cat 01-kubernetes-volumes/static-pvc.yaml
# Copy of ../../02-persistent-storage/pvc.yaml with one change:
# storageClassName: "" so the claim binds to the hand-made PV (student-pv)
# instead of being dynamically provisioned by kind's default "standard" class.
apiVersion: v1
kind: PersistentVolumeClaim
metadata:
  name: student-pvc
spec:
  storageClassName: ""
  volumeName: student-pv
  accessModes:
    - ReadWriteOnce
  resources:
    requests:
      storage: 500Mi

$ kubectl apply -f 01-kubernetes-volumes/static-pvc.yaml
persistentvolumeclaim/student-pvc created

$ kubectl get pv,pvc
NAME                          CAPACITY   ACCESS MODES   RECLAIM POLICY   STATUS   CLAIM                 STORAGECLASS   VOLUMEATTRIBUTESCLASS   REASON   AGE
persistentvolume/student-pv   1Gi        RWO            Retain           Bound    default/student-pvc                  <unset>                          1s

NAME                                STATUS   VOLUME       CAPACITY   ACCESS MODES   STORAGECLASS   VOLUMEATTRIBUTESCLASS   AGE
persistentvolumeclaim/student-pvc   Bound    student-pv   1Gi        RWO                           <unset>                 0s
