### Host-based routing to two backends (my ingress/host-based.yaml)
$ cat ingress/host-based.yaml
# Two tiny backends (hashicorp/http-echo answers with a fixed text) and one Ingress that routes
#   blue.local          -> echo-blue
#   green.local         -> echo-green
#   apps.local/blue     -> echo-blue    (host + path)
#   apps.local/green    -> echo-green
apiVersion: apps/v1
kind: Deployment
metadata:
  name: echo-blue
spec:
  replicas: 1
  selector:
    matchLabels: {app: echo-blue}
  template:
    metadata:
      labels: {app: echo-blue}
    spec:
      containers:
        - name: echo
          image: hashicorp/http-echo:1.0
          args: ["-listen=:5678", "-text=hello from BLUE backend"]
          ports: [{containerPort: 5678}]
          resources: {requests: {cpu: "10m", memory: "8Mi"}}
---
apiVersion: apps/v1
kind: Deployment
metadata:
  name: echo-green
spec:
  replicas: 1
  selector:
    matchLabels: {app: echo-green}
  template:
    metadata:
      labels: {app: echo-green}
    spec:
      containers:
        - name: echo
          image: hashicorp/http-echo:1.0
          args: ["-listen=:5678", "-text=hello from GREEN backend"]
          ports: [{containerPort: 5678}]
          resources: {requests: {cpu: "10m", memory: "8Mi"}}
---
apiVersion: v1
kind: Service
metadata:
  name: echo-blue
spec:
  selector: {app: echo-blue}
  ports: [{port: 80, targetPort: 5678}]
---
apiVersion: v1
kind: Service
metadata:
  name: echo-green
spec:
  selector: {app: echo-green}
  ports: [{port: 80, targetPort: 5678}]
---
apiVersion: networking.k8s.io/v1
kind: Ingress
metadata:
  name: echo-hosts
spec:
  ingressClassName: nginx
  rules:
    - host: blue.local
      http:
        paths:
          - path: /
            pathType: Prefix
            backend: {service: {name: echo-blue, port: {number: 80}}}
    - host: green.local
      http:
        paths:
          - path: /
            pathType: Prefix
            backend: {service: {name: echo-green, port: {number: 80}}}
    - host: apps.local
      http:
        paths:
          - path: /blue
            pathType: Prefix
            backend: {service: {name: echo-blue, port: {number: 80}}}
          - path: /green
            pathType: Prefix
            backend: {service: {name: echo-green, port: {number: 80}}}

### Host + path on one hostname (apps.local/blue, apps.local/green)
$ curl -s -H 'Host: apps.local' http://localhost:30081/blue
hello from BLUE backend

$ curl -s -H 'Host: apps.local' http://localhost:30081/green
hello from GREEN backend

$ curl -s -o /dev/null -w 'apps.local/purple -> HTTP %{http_code}\n' -H 'Host: apps.local' http://localhost:30081/purple
apps.local/purple -> HTTP 404

### 10 requests per host - each host always lands on its own backend
$ sh -c 'for h in blue.local green.local; do for i in $(seq 1 10); do curl -s -H "Host: $h" http://localhost:30081/; done; done | sort | uniq -c'
  10 hello from BLUE backend
  10 hello from GREEN backend
