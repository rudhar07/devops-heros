### Setup: a real PostgreSQL whose password is 'mypassword' (FAKE demo value)
$ kubectl apply -f troubleshooting/base64-gotcha-postgres.yaml
secret/postgres-server-secret created
pod/yatri-postgres created
service/yatri-postgres created

$ kubectl wait --for=condition=Ready pod/yatri-postgres --timeout=240s
pod/yatri-postgres condition met

### BEFORE: the app's Secret was made with 'echo "mypassword" | base64'
$ cat troubleshooting/app-secret-broken.yaml
# BROKEN on purpose: the developer encoded the password with `echo "mypassword" | base64`
# (no -n), so the value carries a trailing newline. FAKE demo value.
apiVersion: v1
kind: Secret
metadata:
  name: app-db-secret
type: Opaque
data:
  DB_PASSWORD: bXlwYXNzd29yZAo=

$ kubectl apply -f troubleshooting/app-secret-broken.yaml
secret/app-db-secret created

$ kubectl apply -f troubleshooting/app-db-client.yaml
pod/app-db-client created

$ kubectl get pod app-db-client
NAME            READY   STATUS   RESTARTS   AGE
app-db-client   0/1     Error    0          5s

$ kubectl logs app-db-client
password length seen by the app: 11 bytes
0000000   m   y   p   a   s   s   w   o   r   d  \n
0000013
psql: error: connection to server at "yatri-postgres" (10.96.3.214), port 5432 failed: FATAL:  password authentication failed for user "yatri_admin"

### Investigate: the password 'looks' right, so look at the bytes
$ sh -c 'kubectl get secret app-db-secret -o jsonpath='"'"'{.data.DB_PASSWORD}'"'"' | base64 -d | xxd'
00000000: 6d79 7061 7373 776f 7264 0a              mypassword.

$ sh -c 'echo "mypassword" | xxd'
00000000: 6d79 7061 7373 776f 7264 0a              mypassword.

$ sh -c 'echo "mypassword" | base64'
bXlwYXNzd29yZAo=

$ sh -c 'echo -n "mypassword" | base64'
LW4gbXlwYXNzd29yZAo=

# Surprise: macOS /bin/sh is bash in POSIX mode, where echo does not understand -n. It printed '-n mypassword' plus a newline:
$ sh -c 'echo -n "mypassword" | xxd'
00000000: 2d6e 206d 7970 6173 7377 6f72 640a       -n mypassword.

# The portable way is printf (or zsh/bash, where echo -n works):
$ sh -c 'printf "%s" "mypassword" | base64'
bXlwYXNzd29yZA==

$ zsh -c 'echo -n "mypassword" | base64'
bXlwYXNzd29yZA==

$ bash -c 'echo -n "mypassword" | base64'
bXlwYXNzd29yZA==

$ kubectl logs yatri-postgres --tail=3
2026-10-07 18:03:20.050 UTC [1] LOG:  database system is ready to accept connections
2026-10-07 18:03:24.640 UTC [84] FATAL:  password authentication failed for user "yatri_admin"
2026-10-07 18:03:24.640 UTC [84] DETAIL:  Connection matched file "/var/lib/postgresql/data/pg_hba.conf" line 128: "host all all all scram-sha-256"

# Root cause: the stored value is 'mypassword\n' (11 bytes, last byte 0a). Postgres compares all 11 bytes -> auth fails.
### Fix: re-encode with echo -n, apply, and recreate the pod (env vars are read at container start)
$ cat troubleshooting/app-secret-fixed.yaml
# FIXED: encoded with `echo -n "mypassword" | base64` - no trailing newline. FAKE demo value.
# (Even simpler: use stringData: {DB_PASSWORD: mypassword} and let Kubernetes encode it.)
apiVersion: v1
kind: Secret
metadata:
  name: app-db-secret
type: Opaque
data:
  DB_PASSWORD: bXlwYXNzd29yZA==

$ kubectl apply -f troubleshooting/app-secret-fixed.yaml
secret/app-db-secret configured

$ sh -c 'kubectl get secret app-db-secret -o jsonpath='"'"'{.data.DB_PASSWORD}'"'"' | base64 -d | xxd'
00000000: 6d79 7061 7373 776f 7264                 mypassword

$ kubectl delete pod app-db-client
pod "app-db-client" deleted from default namespace

$ kubectl apply -f troubleshooting/app-db-client.yaml
pod/app-db-client created
